// Proposed Root-owned replacement business body for ProjectsService::get.
// The existing HTTP/service outer TX constructs Actor and lends CommonSession.
// Scope, data, sensitive GET audit and pending auth stay in that one transaction.
async fn detail_with_session(
    session:&mut crate::approval_common::CommonSession<'_, '_>,project_id:Uuid,
    actor:&crate::contract_support::Actor,
)->Result<Wire,crate::contract_support::Error>{
    use crate::{contract_support::Error as BusinessError,project_action_authorization::{self as scope,ProjectAction,ProjectActionTarget}};
    let proof=scope::require(session,actor,ProjectAction::ReadProjectAnchor,ProjectActionTarget::Project{id:project_id}).await?;
    let full=proof.scope_kind()=="full-project";
    drop(proof);
    if !full{
        // Deliberately no contract/bid/approval/evidence queries on this branch.
        let mut view:Value=sqlx::query_scalar("SELECT jsonb_build_object('scopeKind','creator-first-assignment-only','projectId',p.id,'projectNo',p.project_no,'projectRowVersion',p.row_version,'ownerOrgId',p.owner_org_id,'ownerUserId',NULL,'creatorUserId',o.creator_user_id,'ownershipOriginRowVersion',o.row_version) FROM poms.project p JOIN poms.project_ownership_origin o ON o.project_id=p.id WHERE p.id=$1 AND p.owner_user_id IS NULL AND o.bootstrap_state='available'")
            .bind(project_id).fetch_optional(&mut *session.connection).await?.ok_or_else(||BusinessError::NotFound("项目不存在。".into()))?;
        view["allowedActions"]=json!(if actor.session.user.permissions.iter().any(|p|p.as_str()=="project:write"){vec!["reassign-owner"]}else{vec![]});
        return Ok(Wire::value(view));
    }
    let project=store::find(session.connection,&project_id.to_string()).await?.ok_or_else(||BusinessError::NotFound("项目不存在。".into()))?;
    let users=store::user_names(session.connection,&project.owner_user_id.into_iter().collect::<Vec<_>>()).await?;
    let orgs=store::org_names(session.connection,&project.owner_org_id.into_iter().collect::<Vec<_>>()).await?;
    let contracts=store::contracts(session.connection,project_id).await?;
    let snapshot=store::approval_snapshot(session.connection,project_id).await?;
    let bid=store::current_bid(session.connection,project_id).await?;
    let lead=store::lead(session.connection,project.source_lead_id).await?;
    let latest=contracts.first();
    let mut projected=projection::plan(&ProjectionInput{
        field_package_key:SensitiveFieldPackageKey::ContractFinance,
        fields:vec![StringField{key:"field".into(),raw_value:latest.map(|r|r.signed_amount.clone()),display_text_when_full:latest.map(|r|format!("{} {}",r.signed_amount,r.currency_code)),masked_display_text:None,denied_display_text:None}],
        user:Some(ProjectionUser{sub:actor.id,username:actor.session.user.username.clone(),permissions:actor.session.user.permissions.iter().map(|p|p.as_str().to_owned()).collect()}),
        target_type:"project".into(),target_id:project_id.to_string(),request_context:actor.request_context.clone(),mode_when_unauthorized:None,masked_display_text:None,denied_display_text:None,
    }).map_err(|_|BusinessError::Internal)?;
    if let Some(event)=projected.event{crate::auth::audit::insert_security_event(session.connection,&event).await?;}
    let signed_amount=projected.fields.remove("field").ok_or(BusinessError::Internal)?;
    let approval=match snapshot.as_ref(){
        Some(s)=>json!({"summarySnapshotId":s.id,"summaryPackageKey":s.summary_package_key,"projectionLevel":s.projection_level,"exportPolicy":s.export_policy,"generatedAt":iso(s.generated_at).map_err(|_|BusinessError::Internal)?}),
        None=>json!({"summarySnapshotId":null,"summaryPackageKey":null,"projectionLevel":null,"exportPolicy":null,"generatedAt":null}),
    };
    let stage_id:Option<Uuid>=sqlx::query_scalar("SELECT current_stage_record_id FROM poms.project WHERE id=$1").bind(project_id).fetch_one(&mut *session.connection).await?;
    let mut result=summary(&project).map_err(|_|BusinessError::Internal)?;
    let extra=json!({
        "scopeKind":"full-project","projectRowVersion":project.row_version,"currentStageRecordId":stage_id,
        "ownerName":project.owner_user_id.and_then(|id|users.get(&id)),"ownerOrgName":project.owner_org_id.and_then(|id|orgs.get(&id)),
        "sourceLeadSummary":lead.as_ref().map(|r|json!({"id":r.id,"leadNo":r.lead_no,"leadName":r.lead_name,"customerId":r.customer_id,"customerName":r.customer_name,"status":r.status})),
        "stageSummary":{"currentStage":project.current_stage,"status":project.status,"plannedSignAt":optional_iso(project.planned_sign_at).map_err(|_|BusinessError::Internal)?,"closedAt":optional_iso(project.closed_at).map_err(|_|BusinessError::Internal)?,"closedReason":project.closed_reason,"blockingReasons":stage_reasons(&project)},
        "currentBidSummary":bid.as_ref().map_or_else(||json!({"bidProcessId":null,"bidStatus":"not_configured","resultStatus":null,"tenderNo":null,"bidPackageNo":null,"summary":null}),|r|json!({"bidProcessId":r.id,"bidStatus":r.current_stage,"resultStatus":r.result_status,"tenderNo":r.tender_no,"bidPackageNo":r.bid_package_no,"summary":r.process_summary})),
        "currentContractSummary":{"activeContractCount":contracts.iter().filter(|r|r.status=="active").count(),"latestContractId":latest.map(|r|r.id),"latestContractNo":latest.map(|r|&r.contract_no),"latestContractStatus":latest.map(|r|&r.status),"signedAmountProjection":signed_amount,"currencyCode":latest.map(|r|&r.currency_code),"signedAt":optional_iso(latest.and_then(|r|r.signed_at)).map_err(|_|BusinessError::Internal)?,"currentSnapshotId":latest.and_then(|r|r.current_snapshot_id)},
        "currentApprovalSummary":approval,"currentConfirmationSummary":{"confirmationRecordId":null,"status":"not_configured","requiredCount":0,"confirmedCount":0,"pendingCount":0,"confirmedAt":null},
        "summarySnapshotId":snapshot.as_ref().map(|r|r.id),"projectionLevel":snapshot.as_ref().map(|r|&r.projection_level),"exportPolicy":snapshot.as_ref().map(|r|&r.export_policy),
        "allowedActions":allowed_actions(&project,&actor.session),"generatedAt":iso(clock_now()).map_err(|_|BusinessError::Internal)?,
    });
    result.as_object_mut().ok_or(BusinessError::Internal)?.extend(extra.as_object().ok_or(BusinessError::Internal)?.clone());
    Ok(Wire::value(result))
}
